-- 1. TABELA DE CLIENTES E SEUS VEÍCULOS
CREATE TABLE clientes (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    telefone VARCHAR(30),
    documento VARCHAR(30), -- CPF ou CNPJ
    endereco TEXT,
    moto_modelo VARCHAR(100), -- Ex: Honda Fan 160
    placa VARCHAR(20),       -- Ex: ABC-1D23 (Para consulta de histórico)
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. TABELA DE FORNECEDORES
CREATE TABLE fornecedores (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    telefone VARCHAR(30),
    documento VARCHAR(30),
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. TABELA DE PRODUTOS E PEÇAS
CREATE TABLE produtos (
    id SERIAL PRIMARY KEY,
    codigo_interno VARCHAR(50) UNIQUE NOT NULL, -- Ex: JM-1001 (Código Exclusivo SKU)
    codigo_barras VARCHAR(50),                  -- Opcional (EAN do fabricante)
    nome VARCHAR(200) NOT NULL,
    categoria VARCHAR(100),                     -- Ex: Lubrificantes, Freios, Transmissão
    quantidade INT DEFAULT 0,
    estoque_minimo INT DEFAULT 2,
    preco_custo NUMERIC(10, 2) DEFAULT 0.00,
    preco_venda NUMERIC(10, 2) DEFAULT 0.00,
    garantia_dias INT DEFAULT 90,               -- Prazo padrão de garantia em dias
    fornecedor_id INT REFERENCES fornecedores(id) ON DELETE SET NULL,
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. TABELA DE CAIXA (Abertura e Fechamento de Turno/Dia)
CREATE TABLE caixa_sessoes (
    id SERIAL PRIMARY KEY,
    valor_abertura NUMERIC(10, 2) NOT NULL,
    valor_fechamento NUMERIC(10, 2),
    status VARCHAR(20) DEFAULT 'Aberto',        -- 'Aberto' ou 'Fechado'
    data_abertura TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    data_fechamento TIMESTAMP WITH TIME ZONE
);

-- 5. TABELA DE ORDENS DE SERVIÇO (OS)
CREATE TABLE ordens_servico (
    id SERIAL PRIMARY KEY,
    cliente_id INT REFERENCES clientes(id) ON DELETE SET NULL,
    placa VARCHAR(20),                          -- Placa da moto vinculada
    descricao_servico TEXT NOT NULL,
    valor_total NUMERIC(10, 2) DEFAULT 0.00,
    status VARCHAR(30) DEFAULT 'Aberta',        -- 'Aberta', 'Em Andamento', 'Concluída', 'Entregue'
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 6. TABELA DE VENDAS (Cabeçalho do PDV)
CREATE TABLE vendas (
    id SERIAL PRIMARY KEY,
    cliente_id INT REFERENCES clientes(id) ON DELETE SET NULL,
    caixa_id INT REFERENCES caixa_sessoes(id) ON DELETE SET NULL,
    valor_total NUMERIC(10, 2) NOT NULL,
    forma_pagamento VARCHAR(50) NOT NULL,       -- 'Dinheiro', 'Pix', 'Cartão de débito', 'Cartão de crédito', 'A prazo'
    status_pagamento VARCHAR(30) DEFAULT 'Pago', -- 'Pago' ou 'Pendente' (Fiado)
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 7. ITENS DA VENDA (Peças do Estoque + Mão de Obra Livre)
CREATE TABLE venda_itens (
    id SERIAL PRIMARY KEY,
    venda_id INT REFERENCES vendas(id) ON DELETE CASCADE,
    produto_id INT REFERENCES produtos(id) ON DELETE SET NULL, -- Nulo se for mão de obra livre
    tipo_item VARCHAR(20) NOT NULL,            -- 'produto' ou 'servico_livre'
    descricao_item VARCHAR(200) NOT NULL,       -- Nome da peça ou descrição do serviço
    quantidade INT NOT NULL,
    preco_unitario NUMERIC(10, 2) NOT NULL,
    subtotal NUMERIC(10, 2) NOT NULL,
    data_limite_garantia DATE                   -- Calculado na venda se houver garantia
);

-- 8. TABELA DE REGISTRO / ACIONAMENTO DE GARANTIAS
CREATE TABLE garantias_acionadas (
    id SERIAL PRIMARY KEY,
    venda_item_id INT REFERENCES venda_itens(id) ON DELETE CASCADE,
    cliente_id INT REFERENCES clientes(id) ON DELETE SET NULL,
    produto_id INT REFERENCES produtos(id) ON DELETE SET NULL,
    data_solicitacao TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    motivo_defeito TEXT NOT NULL,
    status VARCHAR(30) DEFAULT 'Em Análise',    -- 'Em Análise', 'Trocado/Aprovado', 'Recusado'
    observacoes TEXT
);

-- 9. TABELA DE CONTAS E FINANCEIRO (A Pagar / A Receber / Fiado)
CREATE TABLE financeiro (
    id SERIAL PRIMARY KEY,
    descricao VARCHAR(200) NOT NULL,
    tipo VARCHAR(20) NOT NULL,                  -- 'Pagar' ou 'Receber'
    venda_id INT REFERENCES vendas(id) ON DELETE CASCADE, -- Vinculado se for venda a prazo (fiado)
    valor NUMERIC(10, 2) NOT NULL,
    vencimento DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'Pendente',      -- 'Pendente', 'Pago', 'Atrasado'
    criado_em TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ========================================================
-- ÍNDICES PARA DESEMPENHO E CONSULTAS RÁPIDAS
-- ========================================================
CREATE INDEX idx_produtos_codigo ON produtos(codigo_interno);
CREATE INDEX idx_clientes_placa ON clientes(placa);
CREATE INDEX idx_vendas_cliente ON vendas(cliente_id);
CREATE INDEX idx_venda_itens_venda ON venda_itens(venda_id);

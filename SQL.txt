-- Habilitar extensão para IDs únicos (UUIDs opcionais ou usar chaves textuais)
create extension if not exists "uuid-ossp";

-- 1. Tabela de Configurações do Negócio
create table if not exists public.settings (
  id uuid default uuid_generate_v4() primary key,
  shop_name text not null default 'Oficina & Peças',
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Insere um registro padrão inicial se estiver vazio
insert into public.settings (shop_name) 
select 'Oficina & Peças'
where not exists (select 1 from public.settings);

-- 2. Tabela de Produtos & Estoque (Código automático PEC-00X)
create table if not exists public.products (
  id text primary key,
  code text not null,
  name text not null,
  price numeric(10,2) not null default 0.00,
  stock integer not null default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Tabela de Clientes
create table if not exists public.customers (
  id text primary key,
  name text not null,
  phone text,
  doc text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 4. Tabela de Ordens de Serviço (OS)
create table if not exists public.orders (
  id text primary key,
  number text not null,
  customer text not null,
  vehicle text not null,
  plate text not null,
  problem text,
  value numeric(10,2) not null default 0.00,
  status text not null default 'Em andamento',
  date date not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 5. Tabela de Histórico de Manutenções por Veículo/Placa (Prontuário)
create table if not exists public.history (
  id text primary key,
  date date not null,
  number text not null,
  customer text not null,
  vehicle text not null,
  plate text not null,
  service text not null,
  value numeric(10,2) not null default 0.00,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 6. Tabela de Vendas (PDV)
create table if not exists public.sales (
  id text primary key,
  date date not null,
  total numeric(10,2) not null default 0.00,
  payment text not null,
  items jsonb not null default '[]'::jsonb,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 7. Configuração de Segurança (Row Level Security - RLS)
-- Como o app usa chave anon e opera local/nuvem direto pelo navegador do balcão, liberamos o acesso básico de leitura/escrita
alter table public.settings enable row level security;
alter table public.products enable row level security;
alter table public.customers enable row level security;
alter table public.orders enable row level security;
alter table public.history enable row level security;
alter table public.sales enable row level security;

-- Políticas de acesso público para operação ágil do sistema local
create policy "Permitir acesso total settings" on public.settings for all using (true) with check (true);
create policy "Permitir acesso total products" on public.products for all using (true) with check (true);
create policy "Permitir acesso total customers" on public.customers for all using (true) with check (true);
create policy "Permitir acesso total orders" on public.orders for all using (true) with check (true);
create policy "Permitir acesso total history" on public.history for all using (true) with check (true);
create policy "Permitir acesso total sales" on public.sales for all using (true) with check (true);
